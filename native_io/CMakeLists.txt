cmake_minimum_required(VERSION 3.16)
project(marvin_policy_native_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(protocol_test protocol_test.cpp)
target_compile_options(protocol_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME protocol_test COMMAND protocol_test)
add_executable(lifecycle_test lifecycle_test.cpp)
target_compile_options(lifecycle_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME lifecycle_test COMMAND lifecycle_test)
add_executable(action_output_test action_output_test.cpp)
target_compile_options(action_output_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME action_output_test COMMAND action_output_test)

option(BUILD_ROS_IO "Build the opt-in ROS I/O helper" OFF)
if(BUILD_ROS_IO)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(control_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(marvin_trace_msgs REQUIRED)
  find_package(runtime_interfaces REQUIRED)
  add_executable(marvin_policy_io main.cpp)
  target_compile_options(marvin_policy_io PRIVATE -Wall -Wextra -Wpedantic -Werror)
  ament_target_dependencies(marvin_policy_io rclcpp sensor_msgs control_msgs std_msgs marvin_trace_msgs runtime_interfaces)
  install(TARGETS marvin_policy_io DESTINATION lib/marvin_policy_server)
endif()

add_executable(gateway_output_test gateway_output_test.cpp)
target_compile_options(gateway_output_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME gateway_output_test COMMAND gateway_output_test)
